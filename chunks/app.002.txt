(0,0,1)),hash(i+vec3(1,0,1)),f.x),mix(hash(i+vec3(0,1,1)),hash(i+vec3(1,1,1)),f.x),f.y),f.z);

function makeEntityMaterial(base, core, variant=0) {
  return new THREE.ShaderMaterial({
    uniforms: {
      uTime: { value: 0 },
      uReaction: { value: 0 },
      uVariant: { value: variant },
      uBase: { value: new THREE.Color(base) },
      uCore: { value: new THREE.Color(core) },
    },
    vertexShader: entityVertex,
    fragmentShader: entityFragment,
  });
}

class Entity {
  constructor({ parent, position = [0, 0, 0], core = '#c9b8a3', base = '#0b0a09', variant = 'ale', scale = 1 } = {}) {
    this.group = new THREE.Group();
    this.group.position.set(...position);
    this.group.scale.setScalar(scale);
    parent.add(this.group);

    this.visual = new THREE.Group();
    this.group.add(this.visual);
    this.reaction = 0;
    this.meet = 0;
    this.variant = variant;
    const variantN = variant === 'kim' ? 1 : 0;

    this.material = makeEntityMaterial(base, core, variantN);
    this.extensionMaterial = makeEntityMaterial('#0a0908', core, variantN + .35);
    this.materials = [this.material, this.extensionMaterial];

    // One dominant continuous sculptural body: a lathed biomorphic monolith,
    // deformed in shader so it never reads as a humanoid assembled from primitives.
    const profile = variant === 'kim'
      ? [[0.10,-1.35],[0.42,-1.22],[0.68,-.82],[0.76,-.22],[0.72,.32],[0.78,.92],[0.63,1.55],[0.43,2.03],[0.26,2.36],[0.11,2.52]]
      : [[0.12,-1.35],[0.48,-1.18],[0.73,-.72],[0.79,-.10],[0.66,.48],[0.82,1.02],[0.58,1.62],[0.46,2.08],[0.23,2.38],[0.09,2.52]];
    const lathePts = profile.map(([r,y]) => new THREE.Vector2(r,y));
    this.body = new THREE.Mesh(new THREE.LatheGeometry(lathePts, 56), this.material);
    this.body.scale.set(1.16, 1.0, .86);
    this.body.rotation.y = variant === 'kim' ? .26 : -.20;
    this.visual.add(this.body);

    // A subtle carved opening hints at luminous matter without becoming a literal heart.
    this.inner = new THREE.Mesh(
      new THREE.SphereGeometry(.42, 24, 18),
      new THREE.MeshBasicMaterial({ color: core, transparent: true, opacity: .018, blending: THREE.AdditiveBlending, depthWrite: false })
    );
    this.inner.position.set(variant === 'kim' ? -.13 : .12, .78, .18);
    this.inner.scale.set(.75, 1.25, .42);
    this.visual.add(this.inner);

    // The contact appendage exists only while matter reaches toward the other entity.
    this.armPivot = new THREE.Group();
    this.armPivot.position.set(variant === 'kim' ? -.48 : .48, .92, .02);
    this.visual.add(this.armPivot);
    this.extension = new THREE.Mesh(new THREE.CapsuleGeometry(.13, 1.22, 10, 18), this.extensionMaterial);
    this.extension.rotation.z = Math.PI / 2;
    this.extension.position.x = variant === 'kim' ? -.72 : .72;
    this.extension.scale.set(.001, .001, .001);
    this.armPivot.add(this.extension);
    this.hand = new THREE.Mesh(new THREE.SphereGeometry(.17, 18, 14), this.extensionMaterial);
    this.hand.position.x = variant === 'kim' ? -1.38 : 1.38;
    this.hand.scale.setScalar(.001);
    this.armPivot.add(this.hand);

    const pCount = 128;
    const pArr = new Float32Array(pCount * 3);
    this.particleBase = [];
    for (let i = 0; i < pCount; i++) {
      const a = Math.random() * Math.PI * 2;
      const r = .34 + Math.random() * .88;
      const p = new THREE.Vector3(Math.cos(a) * r * .52, (Math.random() - .18) * 3.25 + .2, Math.sin(a) * r * .30);
      this.particleBase.push(p);
      pArr.set([p.x,p.y,p.z], i*3);
    }
    this.particleGeo = new THREE.BufferGeometry();
    this.particleGeo.setAttribute('position', new THREE.BufferAttribute(pArr,3));
    this.particleMat = new THREE.PointsMaterial({ color: core, size: .022, transparent: true, opacity: variant === 'ale' ? .11 : .065, depthWrite: false, blending: THREE.AdditiveBlending });
    this.particles = new THREE.Points(this.particleGeo, this.particleMat);
    this.visual.add(this.particles);
  }

  handWorld() { return this.hand.getWorldPosition(new THREE.Vector3()); }

  update(t, reaction = this.reaction, meet = this.meet) {
    this.reaction = THREE.MathUtils.lerp(this.reaction, reaction, .055);
    this.meet = THREE.MathUtils.lerp(this.meet, meet, .06);
    this.materials.forEach((m, idx) => {
      m.uniforms.uTime.value = t * (idx ? .91 : 1);
      m.uniforms.uReaction.value = this.reaction * .82 + this.meet * .52;
    });

    this.visual.position.y = 1.34 + Math.sin(t*.67 + this.group.position.x*.08)*.075;
    this.visual.rotation.z = Math.sin(t*.31 + (this.variant==='kim'?1.2:0))*.035;
    this.visual.rotation.y += Math.sin(t*.23 + this.group.position.z*.03)*.0007;
    const breathe = 1 + Math.sin(t*.88 + (this.variant==='kim'?.8:0))*.018 + this.reaction*.016;
    this.body.scale.set(1.16*breathe, 1/breathe*.998, .86*breathe);
    this.inner.material.opacity = .012 + this.reaction*.055 + this.meet*.12;
    this.inner.scale.set(.75*(1+this.meet*.12),1.25*(1+this.reaction*.08),.42);

    const reach = smoother(this.meet);
    const exScale = Math.max(.001, reach);
    this.extension.scale.set(exScale, exScale, .62 + reach*.72);
    this.hand.scale.setScalar(exScale);
    this.armPivot.rotation.z = (this.variant==='kim' ? .10 : -.10) + (this.variant==='kim' ? -.12 : .12)*reach;
