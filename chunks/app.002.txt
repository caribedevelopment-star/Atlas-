function makeEntityMaterial(base, core) {
  return new THREE.ShaderMaterial({
    uniforms: {
      uTime: { value: 0 },
      uReaction: { value: 0 },
      uBase: { value: new THREE.Color(base) },
      uCore: { value: new THREE.Color(core) },
    },
    vertexShader: entityVertex,
    fragmentShader: entityFragment,
  });
}

class Entity {
  constructor({ parent, position = [0, 0, 0], core = '#d69155', base = '#090909', variant = 'ale', scale = 1 } = {}) {
    this.group = new THREE.Group();
    this.group.position.set(...position);
    this.group.scale.setScalar(scale);
    parent.add(this.group);

    this.visual = new THREE.Group();
    this.group.add(this.visual);
    this.reaction = 0;
    this.meet = 0;
    this.variant = variant;

    this.material = makeEntityMaterial(base, core);
    this.glossMaterial = makeEntityMaterial('#0c0b0b', core);
    this.materials = [this.material, this.glossMaterial];

    const body = new THREE.Mesh(new THREE.IcosahedronGeometry(0.85, 5), this.material);
    body.scale.set(1.12, 1.55, 0.95);
    body.position.set(0, 1.15, 0);

    const belly = new THREE.Mesh(new THREE.IcosahedronGeometry(0.78, 4), this.material);
    belly.scale.set(1.0, 0.92, 0.88);
    belly.position.set(variant === 'kim' ? 0.18 : -0.18, 0.55, 0.02);

    const shoulder = new THREE.Mesh(new THREE.IcosahedronGeometry(0.55, 4), this.glossMaterial);
    shoulder.scale.set(0.95, 0.72, 0.72);
    shoulder.position.set(variant === 'kim' ? -0.18 : 0.18, 1.95, 0.0);

    const lower = new THREE.Mesh(new THREE.IcosahedronGeometry(0.58, 4), this.glossMaterial);
    lower.scale.set(0.84, 0.78, 0.78);
    lower.position.set(variant === 'kim' ? 0.12 : -0.12, -0.15, 0.03);

    this.visual.add(body, belly, shoulder, lower);

    this.armPivot = new THREE.Group();
    this.armPivot.position.set(variant === 'kim' ? -0.75 : 0.75, 1.2, 0.0);
    this.armPivot.rotation.z = variant === 'kim' ? 0.4 : -0.4;
    this.visual.add(this.armPivot);

    this.arm = new THREE.Mesh(new THREE.CapsuleGeometry(0.18, 1.25, 8, 16), this.material);
    this.arm.rotation.z = Math.PI / 2;
    this.arm.position.x = variant === 'kim' ? -0.78 : 0.78;
    this.armPivot.add(this.arm);

    this.hand = new THREE.Mesh(new THREE.IcosahedronGeometry(0.24, 2), this.glossMaterial);
    this.hand.position.x = variant === 'kim' ? -1.52 : 1.52;
    this.armPivot.add(this.hand);

    this.inner = new THREE.Mesh(
      new THREE.IcosahedronGeometry(0.8, 3),
      new THREE.MeshBasicMaterial({ color: core, transparent: true, opacity: 0.04, blending: THREE.AdditiveBlending, depthWrite: false })
    );
    this.inner.position.set(0, 1.0, 0.04);
    this.inner.scale.set(0.85, 1.2, 0.75);
    this.visual.add(this.inner);

    const pCount = 110;
    const pArr = new Float32Array(pCount * 3);
    this.particleBase = [];
    for (let i = 0; i < pCount; i++) {
      const a = Math.random() * Math.PI * 2;
      const r = 0.3 + Math.random() * 1.05;
      const p = new THREE.Vector3(Math.cos(a) * r * 0.6, (Math.random() - 0.2) * 3.4 + 0.9, Math.sin(a) * r * 0.34);
      this.particleBase.push(p);
      pArr.set([p.x, p.y, p.z], i * 3);
    }
    this.particleGeo = new THREE.BufferGeometry();
    this.particleGeo.setAttribute('position', new THREE.BufferAttribute(pArr, 3));
    this.particleMat = new THREE.PointsMaterial({ color: core, size: 0.028, transparent: true, opacity: 0.14, depthWrite: false, blending: THREE.AdditiveBlending });
    this.particles = new THREE.Points(this.particleGeo, this.particleMat);
    this.visual.add(this.particles);
  }

  handWorld() {
    return this.hand.getWorldPosition(new THREE.Vector3());
  }

  update(t, reaction = this.reaction, meet = this.meet) {
    this.reaction = THREE.MathUtils.lerp(this.reaction, reaction, 0.07);
    this.meet = THREE.MathUtils.lerp(this.meet, meet, 0.07);
    this.materials.forEach((m, idx) => {
      m.uniforms.uTime.value = t * (idx ? 0.93 : 1.0);
      m.uniforms.uReaction.value = this.reaction * 0.85 + this.meet * 0.6;
    });

    this.visual.position.y = 0.96 + Math.sin(t * 0.9 + this.group.position.x * 0.08) * 0.12 + this.reaction * 0.03;
    this.visual.rotation.y = Math.sin(t * 0.36 + this.group.position.z * 0.05) * 0.08;
    this.visual.rotation.z = Math.sin(t * 0.55 + this.group.position.x * 0.09) * 0.035;
    this.inner.material.opacity = 0.03 + this.reaction * 0.12 + this.meet * 0.13;
    const pulse = 1 + Math.sin(t * 1.6 + (this.variant === 'kim' ? 0.6 : 0)) * 0.08 + this.reaction * 0.05 + this.meet * 0.1;
    this.inner.scale.set(0.85 * pulse, 1.2 * pulse, 0.75 * pulse);
    this.particleMat.opacity = 0.11 + this.reaction * 0.42 + this.meet * 0.15;

    const armReach = 1 + this.meet * 0.75;
    this.arm.scale.set(1, 1, armReach);
    this.arm.position.x = (this.variant === 'kim' ? -0.78 : 0.78) * (0.8 + this.meet * 0.28);
    this.hand.position.x = (this.variant === 'kim' ? -1.52 : 1.52) * armReach;
    this.armPivot.rotation.z = THREE.MathUtils.lerp(this.armPivot.rotation.z, (this.variant === 'kim' ? 0.38 : -0.38) + (this.variant === 'kim' ? -1 : 1) * this.meet * 0.18, 0.0
