function updateInstancedRise(mesh, source, amount, bob = 0, t = 0) {
  const dummy = crowdDummy;
  for (let i = 0; i < source.length; i++) {
    const d = source[i];
    dummy.position.set(d.pos.x, d.pos.y + amount * d.rise + Math.sin(t * 0.9 + i * 0.17) * bob * amount, d.pos.z);
    dummy.rotation.y = d.rot + amount * 0.35;
    dummy.scale.setScalar(d.scl * (1 + amount * 0.08));
    dummy.updateMatrix();
    mesh.setMatrixAt(i, dummy.matrix);
  }
  mesh.instanceMatrix.needsUpdate = true;
}

// ---------- AUDIO ----------
async function createAudio() {
  const ext = new Audio('/audio/rawayana.mp3');
  ext.loop = true; ext.preload = 'auto'; ext.volume = 0.72;
  try {
    await new Promise((res, rej) => {
      const ok = () => res(); const bad = () => rej();
      ext.addEventListener('canplaythrough', ok, { once: true });
      ext.addEventListener('error', bad, { once: true });
      ext.load(); setTimeout(bad, 1000);
    });
    await ext.play();
    return { set(p) { ext.volume = 0.66 - p * 0.32; }, fade(v) { ext.volume = v; } };
  } catch {
    const AC = window.AudioContext || window.webkitAudioContext;
    if (!AC) return null;
    const ctx = new AC();
    const master = ctx.createGain(); master.gain.value = 0.66; master.connect(ctx.destination);
    const crowdGain = ctx.createGain(); crowdGain.gain.value = 0.14; crowdGain.connect(master);
    const buf = ctx.createBuffer(1, ctx.sampleRate * 2, ctx.sampleRate);
    const arr = buf.getChannelData(0); for (let i = 0; i < arr.length; i++) arr[i] = (Math.random() * 2 - 1) * 0.32;
    const noise = ctx.createBufferSource(); noise.buffer = buf; noise.loop = true;
    const low = ctx.createBiquadFilter(); low.type = 'lowpass'; low.frequency.value = 880;
    noise.connect(low).connect(crowdGain); noise.start();
    const bassGain = ctx.createGain(); bassGain.gain.value = 0.08; bassGain.connect(master);
    const bass = ctx.createOscillator(); bass.type = 'sine'; bass.frequency.value = 52; bass.connect(bassGain); bass.start();
    const pulseGain = ctx.createGain(); pulseGain.gain.value = 0.024; pulseGain.connect(master);
    const pulse = ctx.createOscillator(); pulse.type = 'triangle'; pulse.frequency.value = 104; pulse.connect(pulseGain); pulse.start();
    ctx.resume();
    return {
      set(p) {
        crowdGain.gain.setTargetAtTime(0.14 * (1 - p * 0.92), ctx.currentTime, 0.08);
        bassGain.gain.setTargetAtTime(0.08 * (1 - p * 0.5), ctx.currentTime, 0.08);
        pulseGain.gain.setTargetAtTime(0.024 + p * 0.045, ctx.currentTime, 0.08);
      },
      fade(v) { master.gain.setTargetAtTime(v, ctx.currentTime, 0.15); }
    };
  }
}
