function updateTunnel(t) {
  const p = phaseTime();
  camera.position.set(Math.sin(p * 0.22) * 0.25, Math.sin(p * 0.4) * 0.18, 4 - p * 10.5);
  camera.rotation.z = Math.sin(p * 0.8) * 0.07;
  tunnelRings.forEach((r, i) => {
    r.rotation.z += 0.01 * (1 + (i % 3) * 0.18);
    r.material.opacity = 0.08 + Math.max(0, Math.sin(t * 2 + i * 0.45)) * 0.18;
  });
  tunnelParticles.rotation.z = p * 0.06;
}

function updateMeadowSeasons(t) {
  const zProg = clamp01((14 - meadowAle.group.position.z) / 48);
  const together = 1 - clamp01((meadowAle.group.position.distanceTo(meadowKim.group.position) - 1.8) / 6);
  worldProgress = THREE.MathUtils.lerp(worldProgress, zProg * 0.8 + together * 0.2, 0.03);

  const desert = 1 - smooth(worldProgress * 1.1);
  const winter = smooth((worldProgress - 0.12) / 0.22) * (1 - smooth((worldProgress - 0.43) / 0.18));
  const spring = smooth((worldProgress - 0.38) / 0.22);
  const lavenderPhase = smooth((worldProgress - 0.62) / 0.28);

  dryMat.opacity = 0.2 + desert * 0.42;
  snowMat.opacity = winter * 0.62;
  grassMat.opacity = spring * 0.75;
  lavMat.opacity = lavenderPhase * 0.94;
  meadowDust.material.opacity = 0.06 + desert * 0.22;
  meadowSnow.material.opacity = winter * 0.38;
  meadowWarm.material.opacity = lavenderPhase * 0.34;

  meadowFloorMat.color.lerp(
    worldProgress < 0.15 ? new THREE.Color('#8e755b') : worldProgress < 0.38 ? new THREE.Color('#d9dcdd') : worldProgress < 0.7 ? new THREE.Color('#697f56') : new THREE.Color('#7a9060'),
    0.025
  );
  horizon.material.color.lerp(worldProgress < 0.18 ? new THREE.Color('#c5b9aa') : worldProgress < 0.38 ? new THREE.Color('#dfe5e8') : worldProgress < 0.72 ? new THREE.Color('#c8d0c7') : new THREE.Color('#f3d8b6'), 0.03);
  meadowHemi.intensity = 0.9 + spring * 0.55 + lavenderPhase * 0.4;
  meadowSun.intensity = 1.6 + spring * 0.6 + lavenderPhase * 0.85;
  scene.fog.color.lerp(worldProgress < 0.18 ? new THREE.Color('#b9b3ab') : worldProgress < 0.38 ? new THREE.Color('#dce2e5') : worldProgress < 0.72 ? new THREE.Color('#c7d1c6') : new THREE.Color('#efd2b1'), 0.03);
  scene.background.lerp(worldProgress < 0.18 ? new THREE.Color('#b8b0a7') : worldProgress < 0.38 ? new THREE.Color('#e1e6e8') : worldProgress < 0.72 ? new THREE.Color('#c8d0cb') : new THREE.Color('#f1d5b7'), 0.02);

  if (worldProgress < 0.12) setHint('DEL DESIERTO HACIA EL INVIERNO');
  else if (worldProgress < 0.38) setHint('TODO SE DETIENE');
  else if (worldProgress < 0.7) setHint('EL MUNDO RECUERDA CÓMO ESTAR VIVO');
  else if (worldProgress < 0.94) setHint('LA LAVANDA DESPIERTA');
  else setHint('ACÉRCATE A ELLA PARA ASCENDER');

  return { winter, spring, lavenderPhase };
}

  const close = meadowAle.group.position.distanceTo(meadowKim.group.position) < 2.6;
  if (worldProgress > 0.97 && close) beginAscension();
}
